Хичээл 06 — Node.js сервер

aedes брокер + HTTP статик + камер endpoint

mqtt-server.js нь гурван зүйл нэг процесст хийдэг:

  1. MQTT брокер (aedes) — TCP (1883) + WebSocket (8080)
  2. HTTP сервер — radar.html болон бусад статик файлууд
  3. Камерын frame хадгалагч болон MJPEG stream

function sendFile(res, filePath, contentType) {
  fs.readFile(filePath, function (err, data) {
    if (err) return res.writeHead(404) && res.end('not found');
    res.writeHead(200, { 'Content-Type': contentType });
    res.end(data);
  });
}

2. Портууд: fallback

Порт авч чадахгүй бол (жишээ: өөр апп аль хэдийн 8080 ашиглаж байвал) чөлөөт порт руу цацдаг:

function tryListen(server, preferred, label) {
  return new Promise(function (resolve, reject) {
    let usedFree = false;
    function onListening() { resolve(server.address().port); }
    function onError(err) {
      if (err.code === 'EADDRINUSE' && !usedFree) {
        usedFree = true;
        server.once('error', onError);
        server.listen(0);                 // порт 0 = систем чөлөөт порт өгнө
      } else reject(err);
    }
    server.once('listening', onListening);
    server.once('error', onError);
    server.listen(preferred);
  });
}

Брокер ба HTTP сервер тус бүрдээ унасан чөлөөт портыг /config.json дээр хэвлэдэг. Вэб апп энэ файлыг уншиж, өөрийгөө жинхэнэ порт руу холбоно:

fetch('/config.json').then(r => r.json()).then(cfg => {
  connectMqtt(cfg.host, cfg.mqttTcpPort);
});

4. Роботын register (брокерын publish событ)

aedes.on('publish', function (packet, client) {
  const m = /^robots\/([^/]+)\/(status|radar)$/.exec(packet.topic);
  if (m) {
    const name = m[1];
    let parsed = null;
    try { parsed = JSON.parse(packet.payload.toString()); } catch (e) {}
    if (parsed) {
      const r = robots[name] || (robots[name] = { name, lastSeen: 0, lastStatus: null, lastRadar: null });
      r.lastSeen = Date.now();
      if (m[2] === 'status') r.lastStatus = parsed; else r.lastRadar = parsed;
    }
  }
});

Энэ нь /api/robots эндпоинтэд тохируулагддаг:

6. Хичээлийн файлуудыг сервэрлэх

Энэ хичээлийн хуудаснууд өөрөө энэ сервэрээс ирдэг! /js-lesson/... гэсэн request нь js-lesson/ фолдероос файл уншиж, mime төрлийг нь тохируулан илгээдэг. Тиймээс бүх зүйл нэг серверээр: вэб апп + хичээл + MQTT + камер.
